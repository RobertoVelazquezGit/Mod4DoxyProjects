49 const std::vector<ClassInfo>& classes) {

112 std::string DocumentationGenerator::formatFunction(
113 const CodeAnalyzer::FunctionInfo& func, OutputFormat /*format*/) {
114 // The unused format parameter is unnamed to avoid compiler warnings.
115 std::ostringstream output;
116 output << func.returnType << ' ' << func.name << '(';
117 for (std::size_t i = 0; i < func.parameters.size(); ++i) {
118 if (i != 0) {
119 output << ", ";
120 }
121 output << func.parameters[i].first << ' ' << func.parameters[i].second;
122 }
123 output << ") - " << func.documentation;
124 return output.str();
125 }
