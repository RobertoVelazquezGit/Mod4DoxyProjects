154 template<typename PriceContainer>
155 double MarketDataProcessor::calculateSharpeRatio(
156 const PriceContainer& returns,
157 double riskFreeRate)
158 {
159 // Avoid division by zero if the container is empty.
160 if (returns.empty()) {
161 return 0.0;
162 }
163
164 // Calculate the mean of all returns.
165 double mean = 0.0;
166
167 for (const auto& value : returns) {
168 mean += value;
169 }
170
171 mean /= static_cast<double>(returns.size());
172
173 // riskFreeRate is not used yet in this simplified implementation.
174 (void)riskFreeRate;
175
176 return mean;
177 }
